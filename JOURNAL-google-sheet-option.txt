JOURNAL — OPTIONAL "WRITE FROM ANYWHERE" UPGRADE (read-only for visitors)
=========================================================================

By default the journal reads from the JOURNAL list inside build.js — you
edit that, rebuild, and upload. Zero setup, nothing public can change it.

If you'd rather jot entries from your phone WITHOUT touching code, use a
private Google Sheet as the store. Only YOU can edit the Sheet; the website
only READS it, so there's no override problem.

--------------------------------------------------------------------------
STEP 1 — Make the Sheet
--------------------------------------------------------------------------
Create a Google Sheet with a tab called "journal" and these columns in row 1:
    date | title | tags | body
Add one row per entry. tags = comma-separated (e.g. "trading, tools").
Newest at the top OR bottom — the script sorts by row order; put newest first.

--------------------------------------------------------------------------
STEP 2 — Add the Apps Script (Extensions ▸ Apps Script), paste, Deploy ▸
         New deployment ▸ Web app ▸ Execute as: Me ▸ Access: Anyone.
         Copy the /exec URL it gives you.
--------------------------------------------------------------------------
function doGet(){
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('journal');
  const rows = sh.getDataRange().getValues();
  const head = rows.shift(); // date,title,tags,body
  const out = rows
    .filter(r => r[0] || r[1] || r[3])
    .map(r => ({
      date:  String(r[0]||''),
      title: String(r[1]||''),
      tags:  String(r[2]||'').split(',').map(s=>s.trim()).filter(Boolean),
      body:  String(r[3]||'')
    }));
  return ContentService
    .createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}

--------------------------------------------------------------------------
STEP 3 — In journal.html, replace the rendered .jr-list with an empty
         container and add this script. Paste your /exec URL where shown.
--------------------------------------------------------------------------
<script>
const JOURNAL_URL = "PASTE_YOUR_/exec_URL_HERE";
fetch(JOURNAL_URL)
  .then(r => r.json())
  .then(list => {
    const box = document.getElementById("jrList");
    box.innerHTML = list.map(e => {
      const tags = (e.tags && e.tags.length)
        ? '<div class="jr-tags">' + e.tags.map(t => '<span class="jr-tag">'+t+'</span>').join('') + '</div>'
        : '';
      return '<article class="jr-entry in">'
           +   '<div class="jr-date">'+e.date+'</div>'
           +   '<h2>'+e.title+'</h2>'
           +   tags
           +   '<p>'+e.body+'</p>'
           + '</article>';
    }).join("");
  })
  .catch(() => {
    document.getElementById("jrList").innerHTML =
      '<p style="color:var(--body)">Couldn\u2019t load the journal right now.</p>';
  });
</script>
